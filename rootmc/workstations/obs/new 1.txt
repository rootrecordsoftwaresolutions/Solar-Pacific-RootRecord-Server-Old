<script>
  function update() {
    const now = new Date();
    // Customize the format here
    const options = {
      year: 'numeric', month: '2-digit', day: '2-digit',
      hour: '2-digit', minute: '2-digit', second: '2-digit',
      hour12: false
    };
    document.body.innerText = now.toLocaleString(undefined, options);
  }
  setInterval(update, 1000);
  update();
</script>
